## 1. Rust

```rust
use rusto::download::parse_config_with_auto_download;
use rusto::{DetectTextBuilderExt, DetectTextResult, ImageSource, RustO};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    // 1. Daftarkan backend dan inisialisasi mesin PP-OCRv6 Tiny FP32 (unduh otomatis jika belum ada di cache)
    rusto::init();
    let config = parse_config_with_auto_download("{}")?;
    let mut ocr = RustO::initialize(config)?;

    // 2. Jalankan deteksi baris terstruktur dengan perangkaian fluent builder
    let image = ImageSource::Path("receipt.jpg".into());
    let result = ocr
        .detect_text_builder(image.clone())
        .with_auto_unwarp(true)
        .with_layout_reading(true)
        .as_lines()
        .run()?;

    match result {
        DetectTextResult::Structured(lines) => {
            println!("--- Hasil Baris Terstruktur ({} baris) ---", lines.len());
            for line in lines {
                println!(
                    "[{:5.1}, {:5.1}, {:5.1}, {:5.1}] (Skor: {:.2}) {}",
                    line.frame.left, line.frame.top, line.frame.width, line.frame.height,
                    line.score, line.text
                );
            }
        }
        DetectTextResult::Spatial(text) => println!("{text}"),
    }

    // 3. Jalankan rekonstruksi tata letak visual 2D spasial
    if let DetectTextResult::Spatial(spatial_text) = ocr
        .detect_text_builder(image)
        .as_spatial()
        .run()?
    {
        println!("\n--- Rekonstruksi Tata Letak Spasial ---\n{spatial_text}");
    }

    Ok(())
}
```

## 3. Python

```python
from rusto import RustO

# Inisialisasi mesin dengan preset bawaan PP-OCRv6 Tiny FP32 (unduh otomatis saat pertama dijalankan)
with RustO.initialize() as engine:
    # Deteksi baris terstruktur via fluent builder
    result = (
        engine.detect_text("receipt.jpg")
        .with_auto_unwarp()
        .with_layout_reading()
        .as_lines()
        .run()
    )

    if result.is_STRUCTURED():
        for line in result.results:
            print(f"[{line.score:.2f}] {line.text}")

    # Rekonstruksi tata letak visual 2D spasial
    spatial_result = engine.detect_text("receipt.jpg").as_spatial().run()
    if spatial_result.is_SPATIAL():
        print(spatial_result.text)
```

## 4. Golang

```go
package main

import (
	"fmt"
	"log"

	"github.com/byrizki/rusto-rs/packages/go"
)

func main() {
	// Inisialisasi tanpa konfigurasi (otomatis menyiapkan PP-OCRv6 Tiny FP32)
	engine, err := rusto.Initialize()
	if err != nil {
		log.Fatalf("Gagal inisialisasi: %v", err)
	}
	defer engine.Close()

	// Deteksi baris terstruktur via fluent builder
	res, err := engine.DetectText("receipt.jpg").
		WithAutoUnwarp(true).
		WithLayoutReading(true).
		AsLines().
		Run()
	if err != nil {
		log.Fatalf("OCR gagal: %v", err)
	}

	for _, line := range res.Results {
		fmt.Printf("[%.2f] %s\n", line.Score, line.Text)
	}

	// Rekonstruksi tata letak visual 2D spasial
	spatial, err := engine.DetectText("receipt.jpg").
		AsSpatial().
		Run()
	if err == nil && spatial.IsSpatial {
		fmt.Println("--- Tata Letak Spasial ---")
		fmt.Println(spatial.Text)
	}
}
```

## 5. Java

```java
import com.byrizki.rusto.RustO;
import com.byrizki.rusto.TextResult;

public class App {
    public static void main(String[] args) {
        try (RustO engine = RustO.initialize()) {
            RustO.DetectTextResult structured = engine.detectText("receipt.jpg")
                .withAutoUnwarp()
                .withLayoutReading()
                .asLines()
                .run();

            for (TextResult line : structured.results()) {
                System.out.printf("[%.2f] %s%n", line.score(), line.text());
            }

            RustO.DetectTextResult spatial = engine.detectText("receipt.jpg")
                .asSpatial()
                .run();
            System.out.println("--- Teks Spasial ---");
            System.out.println(spatial.text());
        }
    }
}
```

## 8. React Native

```typescript
import React, { useEffect, useState } from 'react';
import { Button, SafeAreaView, ScrollView, Text } from 'react-native';
import { close, detectText, initialize, TextResult } from '@rustors/react-native';

export default function QuickScanApp() {
  const [lines, setLines] = useState<TextResult[]>([]);

  useEffect(() => {
    initialize();
    return () => {
      close();
    };
  }, []);

  const handleScanStructured = async () => {
    const results = await detectText({ uri: 'file:///path/to/receipt.jpg' })
      .withAutoUnwarp()
      .withLayoutReading()
      .asLines()
      .run();
    setLines(results);
  };

  return (
    <SafeAreaView style={{ flex: 1, padding: 16 }}>
      <Button title="Pindai Baris" onPress={handleScanStructured} />
      <ScrollView style={{ marginTop: 16 }}>
        {lines.map((line, idx) => (
          <Text key={idx}>{line.text} ({Math.round(line.score * 100)}%)</Text>
        ))}
      </ScrollView>
    </SafeAreaView>
  );
}
```

## 10. Android (Kotlin)

```kotlin
package com.example.ocr

import android.content.Context
import com.byrizki.rusto.*

fun processReceipt(context: Context, imageUri: String) {
    RustO.initialize(context).use { ocr ->
        val structured = ocr.detectText(imageUri)
            .withAutoUnwarp()
            .withLayoutReading()
            .asLines()
            .run()

        if (structured is DetectTextResult.Structured) {
            structured.items.forEach { line ->
                println("${line.text} (Skor: ${line.score})")
            }
        }
    }
}
```

## 11. Command Line Interface (CLI)

```bash
# 1. OCR tanpa konfigurasi (otomatis mengunduh model PP-OCRv6 Tiny FP32 saat pertama kali dijalankan)
rusto receipt.jpg

# 2. Keluaran rekonstruksi tata letak teks spasial 2D dengan pelurusan & urutan baca tata letak
rusto --enable-unwarp --enable-layout --format spatial receipt.jpg

# 3. Keluaran tab-separated values (TSV)
rusto --format tsv receipt.jpg

# 4. Pra-unduh aset model untuk lingkungan luring (offline)
rusto download --model ppocrv6-tiny
```
