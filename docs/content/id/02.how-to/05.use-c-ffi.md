RustO! mengekspor C ABI yang stabil melalui pustaka bersama dinamis (`librusto.so`, `librusto.dylib`, `rusto.dll`) serta arsip statis (`librusto.a`, `rusto.lib`), lengkap dengan berkas header `include/rusto.h` yang dihasilkan oleh `cbindgen`.

## Header C (`rusto.h`)

```c
#ifndef RUSTO_H
#define RUSTO_H

#include <stdarg.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

/**
 * Handle instans mesin RustO yang bersifat buram (opaque)
 */
typedef struct ROCRHandle ROCRHandle;

/**
 * Struktur hasil deteksi teks yang kompatibel dengan C ABI
 */
typedef struct {
    char *text;
    float score;
    float box_x1;
    float box_y1;
    float box_x2;
    float box_y2;
    float box_x3;
    float box_y3;
    float box_x4;
    float box_y4;
    float frame_width;
    float frame_height;
    float frame_top;
    float frame_left;
    int region_id;
    int layout_type;
} CTextResult;

#ifdef __cplusplus
extern "C" {
#endif

// 1. Siklus Hidup Mesin
ROCRHandle *rocr_initialize(const char *config_json);
void rocr_free(ROCRHandle *handle);
const char *rocr_version(void);

// 2. Deteksi Teks Terstruktur (output: "lines" atau "words")
int rocr_detect_text_file(ROCRHandle *handle,
                          const char *image_path,
                          const char *options_json,
                          CTextResult **results_out,
                          size_t *count_out);

int rocr_detect_text_data(ROCRHandle *handle,
                          const uint8_t *image_data,
                          size_t image_len,
                          const char *options_json,
                          CTextResult **results_out,
                          size_t *count_out);

int rocr_detect_text_rgb(ROCRHandle *handle,
                         const uint8_t *image_data,
                         size_t image_len,
                         uint32_t width,
                         uint32_t height,
                         size_t stride,
                         const char *options_json,
                         CTextResult **results_out,
                         size_t *count_out);

// 3. Rekonstruksi Tata Letak Spasial 2D (output: "spatial")
char *rocr_detect_text_file_spatial(ROCRHandle *handle,
                                    const char *image_path,
                                    const char *options_json);

char *rocr_detect_text_data_spatial(ROCRHandle *handle,
                                    const uint8_t *image_data,
                                    size_t image_len,
                                    const char *options_json);

char *rocr_detect_text_rgb_spatial(ROCRHandle *handle,
                                   const uint8_t *image_data,
                                   size_t image_len,
                                   uint32_t width,
                                   uint32_t height,
                                   size_t stride,
                                   const char *options_json);

// 4. Dealokasi Memori
void rocr_free_results(CTextResult *results, size_t count);
void rocr_free_string(char *s);

#ifdef __cplusplus
}
#endif

#endif /* RUSTO_H */
```

## Metadata Wilayah & Kode `layout_type` pada `CTextResult`

Ketika analisis tata letak dokumen diaktifkan (`"layoutReading": true`), setiap `CTextResult` memuat indeks wilayah pada `region_id` dan kode kategori semantik pada `layout_type` (bernilai `-1` jika analisis tata letak tidak aktif atau belum ditetapkan):

| Nilai `layout_type` | Kategori Semantik | Deskripsi                                       |
| :-----------------: | :---------------- | :---------------------------------------------- |
|        `-1`         | Tidak ditetapkan  | Analisis tata letak tidak diaktifkan / di luar. |
|         `0`         | `text`            | Paragraf atau blok teks isi dokumen.            |
|         `1`         | `title`           | Judul bab, bagian, atau tajuk utama.            |
|         `2`         | `figure`          | Gambar, ilustrasi, atau bagan.                  |
|         `3`         | `figureCaption`   | Keterangan / takarir gambar.                    |
|         `4`         | `table`           | Area tabel dokumen.                             |
|         `5`         | `tableCaption`    | Judul atau keterangan tabel.                    |
|         `6`         | `header`          | Catatan kepala halaman (_running header_).      |
|         `7`         | `footer`          | Catatan kaki halaman (_footer_).                |
|         `8`         | `reference`       | Daftar pustaka atau sitasi referensi.           |
|         `9`         | `equation`        | Rumus atau persamaan matematika.                |

## Referensi Kode Status (Return Codes)

Fungsi dengan nilai kembalian `int` menggunakan kode status berikut:

| Kode | Konstanta                  | Makna                                                                                   |
| :--: | :------------------------- | :-------------------------------------------------------------------------------------- |
| `0`  | `ROCR_OK`                  | Operasi berhasil.                                                                       |
| `-1` | `ROCR_ERR_NULL_PTR`        | Pointer null diteruskan pada argumen wajib.                                             |
| `-2` | `ROCR_ERR_INVALID_INPUT`   | Path berkas tidak valid, atau dimensi/stride/panjang buffer RGB8 tidak memenuhi syarat. |
| `-3` | `ROCR_ERR_ENGINE`          | Kesalahan internal mesin inferensi atau pemrosesan OCR.                                 |
| `-4` | `ROCR_ERR_INVALID_OPTIONS` | Kesalahan sintaks atau validasi skema JSON `OcrRunOptions`.                             |
| `-5` | `ROCR_ERR_MODE_MISMATCH`   | Nilai `output` pada JSON tidak sesuai dengan varian fungsi C (terstruktur vs spasial).  |

## Contoh Aplikasi C

```c
#include <stdio.h>
#include <stdlib.h>
#include "rusto.h"

int main(void) {
    printf("Versi RustO!: %s\n", rocr_version());

    // 1. Inisialisasi mesin: "{}" otomatis mengunduh model bawaan Indra OCR ke cache lokal.
    // Anda juga dapat menentukan jalur model eksplisit melalui JSON InitializeConfig.
    ROCRHandle *engine = rocr_initialize("{}");
    if (!engine) {
        fprintf(stderr, "Gagal menginisialisasi mesin RustO!.\n");
        return 1;
    }

    // 2. Jalankan deteksi teks terstruktur dengan skema JSON 6-fase v0.4.0
    const char *options_json =
        "{"
        "  \"output\": \"lines\","
        "  \"autoUnwarp\": true,"
        "  \"layoutReading\": true,"
        "  \"postRecognition\": { \"scoreThreshold\": 0.6 }"
        "}";

    CTextResult *results = NULL;
    size_t count = 0;

    int status = rocr_detect_text_file(engine, "receipt.jpg", options_json, &results, &count);
    if (status == 0) {
        printf("Ditemukan %zu baris teks:\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("[%zu] %s (Skor: %.2f) di (%.1f, %.1f) [%.1fx%.1f] region=%d layout=%d\n",
                i, results[i].text, results[i].score,
                results[i].frame_left, results[i].frame_top,
                results[i].frame_width, results[i].frame_height,
                results[i].region_id, results[i].layout_type
            );
        }
        rocr_free_results(results, count);
    } else {
        fprintf(stderr, "Kode kesalahan OCR: %d\n", status);
    }

    // 3. Jalankan pada piksel RGB8 mentah (interleaved, tanpa kanal alfa)
    uint32_t width = 640, height = 480;
    size_t stride = (size_t)width * 3;
    size_t rgb_len = (size_t)(height - 1) * stride + (size_t)width * 3;
    uint8_t *rgb_buf = (uint8_t *)calloc(rgb_len, 1);
    if (rgb_buf) {
        CTextResult *rgb_results = NULL;
        size_t rgb_count = 0;
        if (rocr_detect_text_rgb(engine, rgb_buf, rgb_len, width, height, stride,
                                 "{\"output\":\"lines\"}", &rgb_results, &rgb_count) == 0) {
            rocr_free_results(rgb_results, rgb_count);
        }
        free(rgb_buf);
    }

    // 4. Jalankan rekonstruksi tata letak spasial 2D
    const char *spatial_opts =
        "{"
        "  \"output\": \"spatial\","
        "  \"preRecognition\": { \"lineYThreshold\": 0.5, \"wordXThreshold\": 0.4 }"
        "}";

    char *spatial_text = rocr_detect_text_file_spatial(engine, "receipt.jpg", spatial_opts);
    if (spatial_text) {
        printf("\n--- Tata Letak Spasial ---\n%s\n", spatial_text);
        rocr_free_string(spatial_text);
    }

    // 5. Dealokasi instans mesin
    rocr_free(engine);
    return 0;
}
```

## Kompilasi dan Penautan (Linking)

### Menggunakan GCC / Clang

```bash
gcc main.c -I./include -L./target/release -lrusto -Wl,-rpath,./target/release -o ocr_demo
./ocr_demo
```

### Menggunakan CMake (`CMakeLists.txt`)

```cmake
cmake_minimum_required(VERSION 3.15)
project(OcrApp C)

add_executable(ocr_app main.c)

# Sertakan direktori header rusto.h
target_include_directories(ocr_app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

# Tautkan pustaka bersama librusto
target_link_libraries(ocr_app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/target/release/librusto.so)
```
