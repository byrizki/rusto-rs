CocoaPod `RustO` menyediakan antarmuka Swift native dan XCFramework universal yang mendukung arsitektur Apple Silicon (`arm64`) maupun simulator (`x86_64`, `arm64`).

## Instalasi

Pada file `Podfile` proyek Anda:

```ruby
target 'YourApp' do
  use_frameworks!

  # Runtime mesin OCR inti
  pod 'RustO'

  # Bundel model bawaan (~6 MB)
  pod 'RustO-Models-PPOCRv6-Tiny'
end
```

Kemudian jalankan `pod install`.

## 1. Inisialisasi Mesin

Inisialisasi instance OCR satu kali dan gunakan kembali di berbagai view atau service:

```swift
import Foundation
import RustO

do {
    // 1. Inisialisasi dengan model bawaan default PP-OCRv6
    let ocr = try RustO.initialize()

    // 2. Atau inisialisasi dengan konfigurasi kustom:
    let config = InitializeConfig(
        template: "ppv6",
        detection: DetectionConfig(modelPath: "custom_det.mnn"),
        recognition: RecognitionConfig(
            modelPath: "custom_rec.mnn",
            dictPath: "custom_dict.txt"
        )
    )
    let customOcr = try RustO.initialize(config: config)
} catch {
    print("Gagal inisialisasi OCR: \(error)")
}
```

## 2. Sumber Masukan Gambar

`ImageSource` menerima path berkas sistem, URL, atau buffer byte mentah `Data`:

```swift
import UIKit
import RustO

// 1. Path berkas filesystem atau string URL file
let fileSource = ImageSource.uri("/path/to/receipt.jpg")

// 2. Data JPEG/PNG dari UIImage
guard let image = UIImage(named: "receipt"),
      let imageData = image.jpegData(compressionQuality: 0.9) else { return }

let dataSource = ImageSource.bytes(imageData)
```

## 3. Opsi Permintaan Lokal (`OcrRunOptions`)

Atur konfigurasi per panggilan pemindaian:

```swift
let options = OcrRunOptions(
    output: .lines,            // .lines, .words, atau .spatial
    lineYThreshold: 0.5,       // Toleransi dinamis pengelompokan baris
    wordXThreshold: 0.4,       // Toleransi pemisahan kata/kolom
    textScore: 0.55,           // Batas minimum skor pengenalan
    maxSideLen: 1600,          // Batas dimensi maksimum gambar
    detection: DetectionRunOptions(
        limitSideLen: 960,
        limitType: "max"
    ),
    postprocess: PostprocessRunOptions(
        useDilation: true,     // Dilatasi morfologi untuk struk redup
        unclipRatio: 2.0
    )
)
```

## 4. Contoh Praktis & Penanganan Keluaran

### Contoh A: Deteksi Baris Terstruktur dengan Poligon Batas

```swift
import RustO

func scanDocument(ocr: RustO, source: ImageSource) {
    do {
        let result = try ocr.detectText(source, options: OcrRunOptions(output: .lines))

        if case .structured(let items) = result {
            print("Ditemukan \(items.count) baris teks:")
            for item in items {
                print("Teks: \(item.text) [Kepercayaan: \(String(format: "%.1f%%", item.score * 100))]")
                print("Frame: (X: \(item.frame.left), Y: \(item.frame.top), W: \(item.frame.width), H: \(item.frame.height))")
                print("Poligon: \(item.boxPoints)")
            }
        }
    } catch {
        print("Kesalahan pemindaian OCR: \(error)")
    }
}
```

### Contoh B: Komponen Tampilan Penuh SwiftUI Scanner

```swift
import SwiftUI
import PhotosUI
import RustO

struct DocumentScannerView: View {
    @State private var selectedItem: PhotosPickerItem? = nil
    @State private var selectedImage: UIImage? = nil
    @State private var ocrResult: String = ""
    @State private var isProcessing: Bool = false
    @State private var ocrEngine: RustO? = nil

    var body: some View {
        NavigationView {
            VStack(spacing: 16) {
                if let selectedImage = selectedImage {
                    Image(uiImage: selectedImage)
                        .resizable()
                        .scaledToFit()
                        .frame(maxHeight: 250)
                        .cornerRadius(12)
                }

                PhotosPicker(selection: $selectedItem, matching: .images) {
                    Label("Pilih Foto", systemImage: "photo.on.rectangle")
                        .frame(maxWidth: .infinity)
                        .padding()
                        .background(Color.orange)
                        .foregroundColor(.white)
                        .cornerRadius(10)
                }
                .onChange(of: selectedItem) { newItem in
                    Task {
                        if let data = try? await newItem?.loadTransferable(type: Data.self),
                           let image = UIImage(data: data) {
                            selectedImage = image
                            performOcr(data: data)
                        }
                    }
                }

                if isProcessing {
                    ProgressView("Memproses OCR...")
                }

                ScrollView {
                    Text(ocrResult)
                        .font(.system(.body, design: .monospaced))
                        .padding()
                        .frame(maxWidth: .infinity, alignment: .leading)
                        .background(Color(UIColor.secondarySystemBackground))
                        .cornerRadius(8)
                }
            }
            .padding()
            .navigationTitle("RustO! Pemindai")
            .onAppear {
                ocrEngine = try? RustO.initialize()
            }
        }
    }

    private func performOcr(data: Data) {
        guard let ocr = ocrEngine else { return }
        isProcessing = true
        ocrResult = ""

        DispatchQueue.global(qos: .userInitiated).async {
            let options = OcrRunOptions(
                output: .spatial,
                lineYThreshold: 0.5,
                wordXThreshold: 0.4
            )

            do {
                let result = try ocr.detectText(.bytes(data), options: options)
                DispatchQueue.main.async {
                    if case .spatial(let text) = result {
                        self.ocrResult = text
                    }
                    self.isProcessing = false
                }
            } catch {
                DispatchQueue.main.async {
                    self.ocrResult = "OCR Gagal: \(error.localizedDescription)"
                    self.isProcessing = false
                }
            }
        }
    }
}
```
