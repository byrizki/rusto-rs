Paket `react-native-rusto` menyediakan bridge native berkinerja tinggi dan dukungan TypeScript penuh untuk aplikasi React Native iOS dan Android.

## Instalasi

```bash
npm install react-native-rusto
# atau yarn add react-native-rusto / pnpm add react-native-rusto
```

### Konfigurasi iOS (`ios/Podfile`)

Tambahkan pod model OCR di dalam target aplikasi Anda pada `ios/Podfile`:

```ruby
target 'YourApp' do
  # Tambahkan pod model pilihan:
  pod 'RustO-Models-PPOCRv6-Tiny' # ~6 MB (Direkomendasikan)
end
```

Kemudian jalankan instalasi CocoaPods:

```bash
cd ios && pod install
```

### Konfigurasi Android

Pastikan JitPack sudah terdaftar di root `android/build.gradle` (atau `settings.gradle`):

```groovy
dependencyResolutionManagement {
    repositories {
        google()
        mavenCentral()
        maven { url 'https://jitpack.io' }
    }
}
```

Pada `android/app/build.gradle`, tambahkan dependensi paket model:

```groovy
dependencies {
    implementation 'com.github.byrizki.rusto-rs:rusto-models-ppocrv6-tiny:v0.2.4'
}
```

> [!NOTE]
> Untuk proyek berbasis **Expo**, gunakan _development build_ (`npx expo run:ios` atau `npx expo run:android`) karena kode native C/Rust kustom tidak dapat berjalan pada Expo Go standar.

## 1. Inisialisasi

Panggil `initialize()` satu kali saat komponen aplikasi dimuat:

```typescript
import { initialize } from 'react-native-rusto';

// Menggunakan preset default PP-OCRv6 dan aset model bawaan
await initialize({ preset: 'ppv6' });

// Atau tentukan nama berkas model kustom:
await initialize({
  preset: 'ppv6',
  models: {
    detection: 'custom_det.mnn',
    recognition: 'custom_rec.mnn',
    dictionary: 'custom_dict.txt',
  },
});
```

## 2. Sumber Masukan Gambar

`detectText` menerima objek `ImageSource` yang memuat tepat satu properti valid:

```typescript
import { detectText } from 'react-native-rusto';

// 1. Path filesystem lokal atau file:// URI (termasuk Android content://)
const r1 = await detectText({ uri: 'file:///path/to/receipt.jpg' });

// 2. String encoded Base64 (misal dari react-native-camera atau expo-camera)
const r2 = await detectText({ base64: 'data:image/jpeg;base64,...' });

// 3. Uint8Array atau ArrayBuffer
const r3 = await detectText({ bytes: new Uint8Array(rawBuffer) });
```

## 3. Mode Keluaran & Opsi Deteksi

```typescript
import { detectText, TextResult } from 'react-native-rusto';

// Mode 1: Deteksi baris teks (Default) -> menghasilkan TextResult[]
const lines: TextResult[] = await detectText(
  { uri: imageUri },
  { output: 'lines', textScore: 0.5 }
);

// Mode 2: Deteksi per kata -> menghasilkan TextResult[]
const words: TextResult[] = await detectText(
  { uri: imageUri },
  { output: 'words', wordXThreshold: 0.4 }
);

// Mode 3: Rekonstruksi tata letak spasial -> menghasilkan string
const spatialText: string = await detectText(
  { uri: imageUri },
  { output: 'spatial', lineYThreshold: 0.5, wordXThreshold: 0.4 }
);
```

## 4. Contoh Komponen Pemindai Dokumen React Native

Berikut contoh komponen lengkap dengan pemilihan gambar dari galeri, mode baris terstruktur, dan mode tata letak spasial:

```tsx
import React, { useEffect, useState } from 'react';
import {
  ActivityIndicator,
  Dimensions,
  Image,
  SafeAreaView,
  ScrollView,
  StyleSheet,
  Text,
  TouchableOpacity,
  View,
} from 'react-native';
import { launchImageLibrary } from 'react-native-image-picker';
import { detectText, initialize, TextResult } from 'react-native-rusto';

export default function DocumentScannerScreen() {
  const [imageUri, setImageUri] = useState<string | null>(null);
  const [lines, setLines] = useState<TextResult[]>([]);
  const [spatialText, setSpatialText] = useState<string>('');
  const [loading, setLoading] = useState<boolean>(false);
  const [mode, setMode] = useState<'lines' | 'spatial'>('lines');

  useEffect(() => {
    initialize().catch((err) => console.error('Gagal Inisialisasi RustO:', err));
  }, []);

  const handlePickImage = async () => {
    const result = await launchImageLibrary({ mediaType: 'photo', quality: 0.9 });
    if (result.assets && result.assets[0]?.uri) {
      const uri = result.assets[0].uri;
      setImageUri(uri);
      processOcr(uri);
    }
  };

  const processOcr = async (uri: string) => {
    setLoading(true);
    try {
      if (mode === 'lines') {
        const results = await detectText(
          { uri },
          {
            output: 'lines',
            maxSideLen: 1600,
            postprocess: { useDilation: true },
          }
        );
        setLines(results as TextResult[]);
        setSpatialText('');
      } else {
        const formatted = await detectText(
          { uri },
          {
            output: 'spatial',
            lineYThreshold: 0.5,
            wordXThreshold: 0.4,
          }
        );
        setSpatialText(formatted as string);
        setLines([]);
      }
    } catch (error) {
      console.error('Kesalahan OCR:', error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <SafeAreaView style={styles.container}>
      <View style={styles.header}>
        <TouchableOpacity style={styles.btn} onPress={handlePickImage}>
          <Text style={styles.btnText}>Pilih Gambar</Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={[styles.btn, styles.toggleBtn]}
          onPress={() => {
            const nextMode = mode === 'lines' ? 'spatial' : 'lines';
            setMode(nextMode);
            if (imageUri) processOcr(imageUri);
          }}
        >
          <Text style={styles.btnText}>Mode: {mode.toUpperCase()}</Text>
        </TouchableOpacity>
      </View>

      {loading && <ActivityIndicator size="large" color="#f97316" style={{ marginVertical: 12 }} />}

      <ScrollView style={styles.content}>
        {imageUri && (
          <View style={styles.imageWrapper}>
            <Image source={{ uri: imageUri }} style={styles.previewImage} resizeMode="contain" />
          </View>
        )}

        <View style={styles.resultContainer}>
          {mode === 'spatial' ? (
            <Text style={styles.spatialCode}>{spatialText}</Text>
          ) : (
            lines.map((item, idx) => (
              <View key={idx} style={styles.lineItem}>
                <Text style={styles.lineText}>{item.text}</Text>
                <Text style={styles.lineScore}>({(item.score * 100).toFixed(0)}%)</Text>
              </View>
            ))
          )}
        </View>
      </ScrollView>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#09090b' },
  header: { flexDirection: 'row', padding: 12, gap: 8 },
  btn: { flex: 1, backgroundColor: '#f97316', padding: 12, borderRadius: 8, alignItems: 'center' },
  toggleBtn: { backgroundColor: '#27272a' },
  btnText: { color: '#ffffff', fontWeight: 'bold' },
  content: { flex: 1, padding: 12 },
  imageWrapper: {
    width: '100%',
    height: 250,
    backgroundColor: '#18181b',
    borderRadius: 8,
    marginBottom: 12,
  },
  previewImage: { width: '100%', height: '100%' },
  resultContainer: { padding: 12, backgroundColor: '#18181b', borderRadius: 8 },
  lineItem: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    paddingVertical: 4,
    borderBottomWidth: 1,
    borderBottomColor: '#27272a',
  },
  lineText: { color: '#fafafa', fontSize: 14, flex: 1 },
  lineScore: { color: '#a1a1aa', fontSize: 12, marginLeft: 8 },
  spatialCode: { color: '#38bdf8', fontFamily: 'monospace', fontSize: 13, lineHeight: 18 },
});
```
