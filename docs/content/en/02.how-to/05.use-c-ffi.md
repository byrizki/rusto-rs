RustO! exports a stable C ABI (`include/rusto.h`) through dynamic libraries (`librusto.so`, `librusto.dylib`, `rusto.dll`) and static archives (`librusto.a`, `rusto.lib`).

## C Header (`include/rusto.h`)

Always include `include/rusto.h` generated by `cbindgen` so struct layouts match the compiled binary ABI:

```c
#ifndef RUSTO_H
#define RUSTO_H

#include <stdarg.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct ROCRHandle ROCRHandle;

typedef struct {
    char *text;
    float score;
    float box_x1;
    float box_y1;
    float box_x2;
    float box_y2;
    float box_x3;
    float box_y3;
    float box_x4;
    float box_y4;
    float frame_width;
    float frame_height;
    float frame_top;
    float frame_left;
    int region_id;
    int layout_type;
} CTextResult;

#ifdef __cplusplus
extern "C" {
#endif

// 1. Engine Lifecycle
ROCRHandle *rocr_initialize(const char *config_json);
void rocr_free(ROCRHandle *handle);
const char *rocr_version(void);

// 2. Structured Text Detection (Lines / Words)
int rocr_detect_text_file(ROCRHandle *handle,
                          const char *image_path,
                          const char *options_json,
                          CTextResult **results_out,
                          size_t *count_out);

int rocr_detect_text_data(ROCRHandle *handle,
                          const uint8_t *image_data,
                          size_t image_len,
                          const char *options_json,
                          CTextResult **results_out,
                          size_t *count_out);

int rocr_detect_text_rgb(ROCRHandle *handle,
                         const uint8_t *image_data,
                         size_t image_len,
                         uint32_t width,
                         uint32_t height,
                         size_t stride,
                         const char *options_json,
                         CTextResult **results_out,
                         size_t *count_out);

// 3. Spatial Layout Text Detection
char *rocr_detect_text_file_spatial(ROCRHandle *handle,
                                    const char *image_path,
                                    const char *options_json);

char *rocr_detect_text_data_spatial(ROCRHandle *handle,
                                    const uint8_t *image_data,
                                    size_t image_len,
                                    const char *options_json);

char *rocr_detect_text_rgb_spatial(ROCRHandle *handle,
                                   const uint8_t *image_data,
                                   size_t image_len,
                                   uint32_t width,
                                   uint32_t height,
                                   size_t stride,
                                   const char *options_json);

// 4. Memory Deallocation
void rocr_free_results(CTextResult *results, size_t count);
void rocr_free_string(char *s);

#ifdef __cplusplus
}
#endif

#endif /* RUSTO_H */
```

### `CTextResult` Layout Metadata Fields

When document layout detection is enabled, `CTextResult` populates `region_id` and `layout_type`:

- **`region_id`**: `0`-based index of the enclosing layout block, or `-1` when layout analysis is not active or unassigned.
- **`layout_type`**: Integer discriminant (`-1` when unassigned):
  - `0` = `Text`, `1` = `Title`, `2` = `Figure`, `3` = `FigureCaption`, `4` = `Table`, `5` = `TableCaption`, `6` = `Header`, `7` = `Footer`, `8` = `Reference`, `9` = `Equation`.

## Return Codes Reference

Functions returning `int` status use the following return codes:

| Code | Constant                   | Meaning                                                                             |
| :--- | :------------------------- | :---------------------------------------------------------------------------------- |
| `0`  | `ROCR_OK`                  | Operation succeeded.                                                                |
| `-1` | `ROCR_ERR_NULL_PTR`        | Null pointer passed for required handle or parameter.                               |
| `-2` | `ROCR_ERR_INVALID_INPUT`   | Invalid file path or invalid RGB dimensions / stride / buffer length.               |
| `-3` | `ROCR_ERR_ENGINE`          | Internal OCR pipeline or neural inference error.                                    |
| `-4` | `ROCR_ERR_INVALID_OPTIONS` | `options_json` syntax error or unknown/invalid 6-phase option key.                  |
| `-5` | `ROCR_ERR_MODE_MISMATCH`   | `output` granularity in `options_json` does not match the called C function family. |

## Example C Application

`rocr_initialize("{}")` automatically downloads and caches the default **PP-OCRv6 Tiny FP32** (`ppocrv6-tiny`, Apache 2.0) models on first run, and `options_json` uses the v0.4.0 6-phase JSON schema:

```c
#include <stdio.h>
#include <stdlib.h>
#include "rusto.h"

int main(void) {
    printf("RustO! Version: %s\n", rocr_version());

    // 1. Zero-config initialization (auto-downloads ppocrv6-tiny if not cached)
    ROCRHandle *engine = rocr_initialize("{}");
    if (!engine) {
        fprintf(stderr, "Failed to initialize RustO! engine.\n");
        return 1;
    }

    // 2. Structured OCR detection using v0.4 6-phase options JSON
    const char *options_json =
        "{"
        "  \"output\": \"lines\","
        "  \"autoUnwarp\": true,"
        "  \"layoutReading\": true,"
        "  \"postRecognition\": { \"scoreThreshold\": 0.7 }"
        "}";

    CTextResult *results = NULL;
    size_t count = 0;

    int status = rocr_detect_text_file(engine, "receipt.jpg", options_json, &results, &count);
    if (status == 0) {
        printf("Detected %zu text blocks:\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("[%zu] %s (Score: %.2f, Region: %d, Layout: %d) at (%.1f, %.1f) [%.1fx%.1f]\n",
                i, results[i].text, results[i].score,
                results[i].region_id, results[i].layout_type,
                results[i].frame_left, results[i].frame_top,
                results[i].frame_width, results[i].frame_height
            );
        }
        rocr_free_results(results, count);
    } else {
        fprintf(stderr, "OCR Error code: %d\n", status);
    }

    // 3. Spatial layout reconstruction
    const char *spatial_opts =
        "{"
        "  \"output\": \"spatial\","
        "  \"preRecognition\": { \"lineYThreshold\": 0.6, \"wordXThreshold\": 0.4 }"
        "}";
    char *spatial_text = rocr_detect_text_file_spatial(engine, "receipt.jpg", spatial_opts);
    if (spatial_text) {
        printf("\n--- Spatial Layout ---\n%s\n", spatial_text);
        rocr_free_string(spatial_text);
    }

    // 4. Decoded RGB8 buffer example (interleaved R,G,B, 3 bytes per pixel)
    uint32_t width = 640, height = 480;
    size_t stride = (size_t)width * 3;
    size_t rgb_len = stride * (size_t)height;
    uint8_t *rgb_buf = (uint8_t *)calloc(rgb_len, 1);
    if (rgb_buf) {
        CTextResult *rgb_results = NULL;
        size_t rgb_count = 0;
        if (rocr_detect_text_rgb(engine, rgb_buf, rgb_len, width, height, stride,
                                 "{\"output\":\"lines\"}", &rgb_results, &rgb_count) == 0) {
            rocr_free_results(rgb_results, rgb_count);
        }
        free(rgb_buf);
    }

    // 5. Free engine instance
    rocr_free(engine);
    return 0;
}
```

## Building and Linking

### Using GCC / Clang

```bash
gcc main.c -Iinclude -Ltarget/release -lrusto -Wl,-rpath,target/release -o ocr_demo
./ocr_demo
```

### Using CMake (`CMakeLists.txt`)

```cmake
cmake_minimum_required(VERSION 3.15)
project(OcrApp C)

add_executable(ocr_app main.c)

target_include_directories(ocr_app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(ocr_app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/target/release/librusto.so)
```
