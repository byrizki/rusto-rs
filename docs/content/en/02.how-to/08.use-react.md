# Use React

The `@rustors/react` package provides declarative React primitives: `<RustoProvider />`, `useOcr()`, `<OcrOverlay />`, and `<OcrDropzone />`.

## Installation

```bash
npm install @rustors/react @rustors/web
# or
yarn add @rustors/react @rustors/web
# or
pnpm add @rustors/react @rustors/web
```

## Setup Provider

Wrap your application in `RustoProvider` to manage engine lifecycle and model setup. You can supply models either from local assets or remote URLs:

```tsx
import React from 'react';
import { RustoProvider } from '@rustors/react';

// Option A: Local assets hosted in public/ folder
const localModels = {
  detection: '/models/ppocrv6-tiny/det.rten',
  recognition: '/models/ppocrv6-tiny/rec.rten',
  dictionary: '/models/ppocrv6-tiny/dict.txt',
};

// Option B: Hosted via GitHub Releases or CDN
const RELEASE_BASE = 'https://github.com/byrizki/rusto-rs/releases/download/models-rten-v0.3.2';
const remoteModels = {
  detection: `${RELEASE_BASE}/ppocrv6_tiny_det.rten`,
  recognition: `${RELEASE_BASE}/ppocrv6_tiny_rec.rten`,
  dictionary: `${RELEASE_BASE}/ppocrv6_dict.txt`,
};

export function App() {
  return (
    <RustoProvider
      config={{
        preset: 'ppv6',
        wasmUrl: 'https://cdn.jsdelivr.net/npm/@rustors/web/dist/wasm/rusto_rten_wasm_bg.wasm',
        models: localModels,
      }}
    >
      <OcrScanner />
    </RustoProvider>
  );
}
```

## Scanning with `useOcr` and `<OcrOverlay />`

```tsx
import React, { useState } from 'react';
import { useOcr, OcrOverlay, OcrDropzone } from '@rustors/react';

export function OcrScanner() {
  const [imageSrc, setImageSrc] = useState<string | null>(null);
  const { detect, results, isProcessing } = useOcr();

  const handleFileSelect = async (file: File) => {
    setImageSrc(URL.createObjectURL(file));
    await detect(file);
  };

  return (
    <div>
      <OcrDropzone onFileSelect={handleFileSelect} />

      {isProcessing && <p>Scanning document...</p>}

      {imageSrc && results.length > 0 && (
        <OcrOverlay
          imageSrc={imageSrc}
          results={results}
          onItemClick={(item) => console.log('Clicked box:', item.text)}
        />
      )}
    </div>
  );
}
```

## Content Security Policy (CSP) Troubleshooting

When building React or Next.js applications in enterprise environments with strict CSP headers, ensure these rules are added:

```http
Content-Security-Policy:
  script-src 'self' 'wasm-unsafe-eval' https://cdn.jsdelivr.net;
  connect-src 'self' https://cdn.jsdelivr.net blob:;
  worker-src 'self' blob:;
  img-src 'self' data: blob:;
```

- If you see `EvalError: Refused to compile or instantiate WebAssembly module`, add `'wasm-unsafe-eval'` to `script-src`.
- If model assets fail to download from jsDelivr, add `https://cdn.jsdelivr.net` to `connect-src`.
- If worker instantiation throws a `SecurityError`, add `worker-src 'self' blob:`.
