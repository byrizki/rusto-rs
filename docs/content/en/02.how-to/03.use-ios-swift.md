RustO! provides first-class iOS and macOS support via a universal XCFramework and idiomatic Swift API (`RustO`). It supports Apple Silicon (`arm64`), Intel (`x86_64`), and iOS Simulators with zero C++ toolchain requirements.

---

## Installation

You can integrate RustO! into your iOS project using either **Swift Package Manager (SPM)** or **CocoaPods**.

### Method A: Swift Package Manager (Recommended)

#### In Xcode

1. Open your project in Xcode.
2. Navigate to **File > Add Package Dependencies...**
3. Enter the repository URL:
   ```text
   https://github.com/byrizki/rusto-rs.git
   ```
4. Select `RustO` and set the dependency rule to your preferred version (e.g., `0.4.0`).

#### In `Package.swift`

```swift
// swift-tools-version: 5.9
import PackageDescription

let package = Package(
    name: "MyScannerApp",
    platforms: [
        .iOS(.v13),
        .macOS(.v11)
    ],
    dependencies: [
        .package(url: "https://github.com/byrizki/rusto-rs.git", from: "0.4.0")
    ],
    targets: [
        .target(
            name: "MyScannerApp",
            dependencies: [
                .product(name: "RustO", package: "rusto-rs")
            ]
        )
    ]
)
```

---

### Method B: CocoaPods

In your project's `Podfile`:

```ruby
target 'YourApp' do
  use_frameworks!

  # Core OCR engine runtime XCFramework
  pod 'RustO', '~> 0.4.0'
end
```

Then install the dependencies:

```bash
pod install
```

## 1. Permissions Configuration (`Info.plist`)

If capturing images from the camera or photo library, declare the appropriate usage descriptions in your app's `Info.plist`:

```xml
<key>NSCameraUsageDescription</key>
<string>Allow access to the camera to scan documents and extract text.</string>
<key>NSPhotoLibraryUsageDescription</key>
<string>Allow access to photo library to select document images for OCR.</string>
```

## 2. Engine Initialization & Model Resolution

### Option 1: Zero-Config Auto-Download (`indraocr`)

Calling `RustO.initialize()` with no arguments automatically resolves cached models or downloads the canonical **Indra OCR** models from Hugging Face Hub on first run:

```swift
import RustO

do {
    let ocr = try RustO.initialize()
} catch {
    print("OCR initialization failed: \(error)")
}
```

### Option 2: Bundled Models or Explicit Presets

Load `.mnn` or `.rten` files bundled inside your app target using `initializeWithBundledModels` or `InitializeConfig.indra(...)` / `.ppv6(...)`:

```swift
import Foundation
import RustO

// Load det.mnn (or det.rten), rec.mnn (or rec.rten), and dict.txt (or meta.json) from Bundle.main
let bundledOcr = try RustO.initializeWithBundledModels()

// Or pass explicit paths via InitializeConfig.indra(...)
func createCustomEngine() throws -> RustO {
    guard let detPath = Bundle.main.path(forResource: "indra_det", ofType: "mnn") ?? Bundle.main.path(forResource: "indra_det", ofType: "rten"),
          let recPath = Bundle.main.path(forResource: "indra_rec_latin", ofType: "mnn") ?? Bundle.main.path(forResource: "indra_rec_latin", ofType: "rten"),
          let dictPath = Bundle.main.path(forResource: "indra_rec_latin_meta", ofType: "json") else {
        throw NSError(domain: "RustO", code: -1, userInfo: [NSLocalizedDescriptionKey: "Model files missing from bundle"])
    }

    let config = InitializeConfig.indra(
        detModelPath: detPath,
        recModelPath: recPath,
        dictPath: dictPath
    )
    return try RustO.initialize(config: config)
}
```

## 3. Input Sources

Pass filesystem paths, encoded `Data` buffers, `ImageSource` enums, or decoded RGB8 pixels:

```swift
import UIKit
import RustO

// 1. Local filesystem path
let r1 = try ocr.detectText(image: "/path/to/invoice.png").asLines().run()

// 2. UIImage converted to JPEG or PNG Data
if let image = UIImage(named: "receipt"),
   let imageData = image.jpegData(compressionQuality: 0.95) {
    let r2 = try ocr.detectText(data: imageData).asLines().run()
}

// 3. Decoded interleaved RGB8 pixel buffer
let r3 = try ocr.detectText()
    .rgb(rgbData, width: width, height: height, stride: UInt64(width * 3))
    .asLines()
    .run()
```

## 4. Fluent Builder Chaining (`DetectTextBuilder`)

Fine-tune recognition parameters per request using fluent chaining ending in `.run()`:

```swift
let result = try ocr.detectText(data: imageData)
    .withAutoUnwarp()
    .withLayoutReading()
    .withPostDetection {
        $0.useDilation = true
        $0.unclipRatio = 1.8
    }
    .withPostRecognition {
        $0.scoreThreshold = 0.6
    }
    .asLines()
    .run()
```

## 5. Structured vs Spatial Recognition

### Structured Line Detection (`.asLines()`)

Returns coordinates, polygon vertices, confidence score, and optional layout metadata (`regionId`, `layoutType`) for each line:

```swift
let result = try ocr.detectText(data: imageData)
    .asLines()
    .run()

if case .structured(let items) = result {
    for line in items {
        print("[\(String(format: "%.2f", line.score))] \(line.text)")
        print("  Frame: X=\(line.frame.left), Y=\(line.frame.top), W=\(line.frame.width), H=\(line.frame.height)")
        print("  Polygon Quads: \(line.boxPoints)")
    }
}
```

### Spatial Document Reconstruction (`.asSpatial()`)

Reconstructs multi-column document geometry into aligned plain text:

```swift
let result = try ocr.detectText(data: imageData)
    .asSpatial()
    .run()

if case .spatial(let layoutText) = result {
    print("--- Spatial Layout ---\n\(layoutText)")
}
```

## 6. Complete SwiftUI Scanner Implementation

Offload OCR execution from `@MainActor` using Swift Concurrency (`Task.detached`):

```swift
import SwiftUI
import PhotosUI
import RustO

struct DocumentScannerView: View {
    @State private var selectedItem: PhotosPickerItem?
    @State private var previewImage: UIImage?
    @State private var recognizedText: String = ""
    @State private var isProcessing: Bool = false
    @State private var ocrEngine: RustO?

    var body: some View {
        NavigationStack {
            VStack(spacing: 16) {
                if let previewImage {
                    Image(uiImage: previewImage)
                        .resizable()
                        .scaledToFit()
                        .frame(maxHeight: 240)
                        .cornerRadius(12)
                }

                PhotosPicker(selection: $selectedItem, matching: .images) {
                    Label("Choose Document", systemImage: "doc.text.viewfinder")
                        .font(.headline)
                        .frame(maxWidth: .infinity)
                        .padding()
                        .background(Color.orange)
                        .foregroundColor(.white)
                        .cornerRadius(10)
                }
                .onChange(of: selectedItem) { _, newItem in
                    Task {
                        guard let data = try? await newItem?.loadTransferable(type: Data.self),
                              let uiImage = UIImage(data: data) else { return }
                        previewImage = uiImage
                        await runOcr(data: data)
                    }
                }

                if isProcessing {
                    ProgressView("Extracting text with RustO!...")
                        .padding()
                }

                ScrollView {
                    Text(recognizedText.isEmpty ? "No text recognized yet." : recognizedText)
                        .font(.system(.body, design: .monospaced))
                        .frame(maxWidth: .infinity, alignment: .leading)
                        .padding()
                        .background(Color(UIColor.secondarySystemBackground))
                        .cornerRadius(8)
                }
            }
            .padding()
            .navigationTitle("RustO! iOS Scanner")
            .task {
                ocrEngine = try? RustO.initialize()
            }
        }
    }

    private func runOcr(data: Data) async {
        guard let engine = ocrEngine else { return }
        isProcessing = true
        recognizedText = ""

        let resultText: String = await Task.detached(priority: .userInitiated) {
            do {
                let result = try engine.detectText(data: data)
                    .withAutoUnwarp()
                    .withPreRecognition {
                        $0.lineYThreshold = 0.5
                        $0.wordXThreshold = 0.4
                    }
                    .asSpatial()
                    .run()
                if case .spatial(let text) = result {
                    return text
                }
                return "No text extracted."
            } catch {
                return "OCR Error: \(error.localizedDescription)"
            }
        }.value

        recognizedText = resultText
        isProcessing = false
    }
}
```
