The `RustO` CocoaPod provides a native Swift wrapper and a universal XCFramework supporting Apple Silicon devices (`arm64`) and simulators (`x86_64`, `arm64`).

## Installation

In your `Podfile`:

```ruby
target 'YourApp' do
  use_frameworks!

  # Core OCR engine runtime
  pod 'RustO'

  # Pre-packaged model bundle (~6 MB)
  pod 'RustO-Models-PPOCRv6-Tiny'
end
```

Then run `pod install`.

---

## 1. Engine Initialization

Initialize the OCR instance once and reuse it across views or services:

```swift
import Foundation
import RustO

do {
    // 1. Initialize with default bundled PP-OCRv6 models
    let ocr = try RustO.initialize()

    // 2. Or initialize with custom configuration:
    let config = InitializeConfig(
        template: "ppv6",
        detection: DetectionConfig(modelPath: "custom_det.mnn"),
        recognition: RecognitionConfig(
            modelPath: "custom_rec.mnn",
            dictPath: "custom_dict.txt"
        )
    )
    let customOcr = try RustO.initialize(config: config)
} catch {
    print("Initialization failed: \(error)")
}
```

## 2. Input Sources

`ImageSource` supports file paths, URLs, or raw in-memory `Data`:

```swift
import UIKit
import RustO

// 1. Filesystem path or file URL string
let fileSource = ImageSource.uri("/path/to/receipt.jpg")

// 2. Raw JPEG/PNG Data from UIImage
guard let image = UIImage(named: "receipt"),
      let imageData = image.jpegData(compressionQuality: 0.9) else { return }

let dataSource = ImageSource.bytes(imageData)
```

## 3. Request-Local Options (`OcrRunOptions`)

Configure options per scan call:

```swift
let options = OcrRunOptions(
    output: .lines,            // .lines, .words, or .spatial
    lineYThreshold: 0.5,       // Dynamic row grouping tolerance
    wordXThreshold: 0.4,       // Dynamic column separation tolerance
    textScore: 0.55,           // Minimum recognition score cutoff
    maxSideLen: 1600,          // Maximum image dimension bound
    detection: DetectionRunOptions(
        limitSideLen: 960,
        limitType: "max"
    ),
    postprocess: PostprocessRunOptions(
        useDilation: true,     // Morphological dilation for faint receipts
        unclipRatio: 2.0
    )
)
```

## 4. Practical Recipes & Output Handling

### Recipe A: Structured Line Detection with Bounding Quads

```swift
import RustO

func scanDocument(ocr: RustO, source: ImageSource) {
    do {
        let result = try ocr.detectText(source, options: OcrRunOptions(output: .lines))

        if case .structured(let items) = result {
            print("Detected \(items.count) text lines:")
            for item in items {
                print("Text: \(item.text) [Confidence: \(String(format: "%.1f%%", item.score * 100))]")
                print("Frame: (X: \(item.frame.left), Y: \(item.frame.top), W: \(item.frame.width), H: \(item.frame.height))")
                print("Quad: \(item.boxPoints)")
            }
        }
    } catch {
        print("OCR scanning error: \(error)")
    }
}
```

### Recipe B: Complete SwiftUI Scanner View

```swift
import SwiftUI
import PhotosUI
import RustO

struct DocumentScannerView: View {
    @State private var selectedItem: PhotosPickerItem? = nil
    @State private var selectedImage: UIImage? = nil
    @State private var ocrResult: String = ""
    @State private var isProcessing: Bool = false
    @State private var ocrEngine: RustO? = nil

    var body: some View {
        NavigationView {
            VStack(spacing: 16) {
                if let selectedImage = selectedImage {
                    Image(uiImage: selectedImage)
                        .resizable()
                        .scaledToFit()
                        .frame(maxHeight: 250)
                        .cornerRadius(12)
                }

                PhotosPicker(selection: $selectedItem, matching: .images) {
                    Label("Select Photo", systemImage: "photo.on.rectangle")
                        .frame(maxWidth: .infinity)
                        .padding()
                        .background(Color.orange)
                        .foregroundColor(.white)
                        .cornerRadius(10)
                }
                .onChange(of: selectedItem) { newItem in
                    Task {
                        if let data = try? await newItem?.loadTransferable(type: Data.self),
                           let image = UIImage(data: data) {
                            selectedImage = image
                            performOcr(data: data)
                        }
                    }
                }

                if isProcessing {
                    ProgressView("Processing OCR...")
                }

                ScrollView {
                    Text(ocrResult)
                        .font(.system(.body, design: .monospaced))
                        .padding()
                        .frame(maxWidth: .infinity, alignment: .leading)
                        .background(Color(UIColor.secondarySystemBackground))
                        .cornerRadius(8)
                }
            }
            .padding()
            .navigationTitle("RustO! Scanner")
            .onAppear {
                ocrEngine = try? RustO.initialize()
            }
        }
    }

    private func performOcr(data: Data) {
        guard let ocr = ocrEngine else { return }
        isProcessing = true
        ocrResult = ""

        DispatchQueue.global(qos: .userInitiated).async {
            let options = OcrRunOptions(
                output: .spatial,
                lineYThreshold: 0.5,
                wordXThreshold: 0.4
            )

            do {
                let result = try ocr.detectText(.bytes(data), options: options)
                DispatchQueue.main.async {
                    if case .spatial(let text) = result {
                        self.ocrResult = text
                    }
                    self.isProcessing = false
                }
            } catch {
                DispatchQueue.main.async {
                    self.ocrResult = "OCR Failed: \(error.localizedDescription)"
                    self.isProcessing = false
                }
            }
        }
    }
}
```
