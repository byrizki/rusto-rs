The Android SDK (`com.byrizki.rusto`) is distributed via JitPack as an AAR containing pre-compiled JNI shared libraries for `arm64-v8a`, `armeabi-v7a`, `x86`, and `x86_64`.

## Installation

Add JitPack to your root `settings.gradle`:

```groovy
dependencyResolutionManagement {
    repositories {
        google()
        mavenCentral()
        maven { url 'https://jitpack.io' }
    }
}
```

In your app module's `build.gradle`:

```groovy
dependencies {
    // Core OCR runtime with JNI native binaries
    implementation 'com.github.byrizki.rusto-rs:rusto-android:v0.4.1'
}
```

## 1. Engine Lifecycle & Asset Resolution

Passing an Android `Context` to `RustO.initialize(context)` automatically extracts bundled `.mnn` / `.rten` assets from `assets/` if present, or downloads **PP-OCRv6 Tiny FP32** (`ppocrv6-tiny`, Apache 2.0) on demand from Hugging Face Hub into `context.cacheDir/rusto/models`:

```kotlin
package com.example.ocr

import android.content.Context
import com.byrizki.rusto.*

class OcrManager(private val context: Context) : AutoCloseable {
    // 1. Zero-config initialization (extracts bundled assets or auto-downloads ppocrv6-tiny)
    private val ocr: RustO = RustO.initialize(context)

    // 2. Or initialize with custom configuration / bundled assets:
    fun initCustom(): RustO {
        val config = InitializeConfig(
            model = "ppocrv6-tiny",
            enableReadingOrder = true,
            unwarp = UnwarpConfig(enabled = true),
            layout = LayoutConfig(enabled = true)
        )
        return RustO.initialize(context, config)
    }

    override fun close() {
        ocr.close()
    }
}
```

## 2. Supported Image Sources

`detectText` resolves Android `content://` URIs, `file://` URIs, local filesystem paths, encoded `ByteArray` buffers, and decoded RGB8 byte arrays:

```kotlin
// 1. Content URI from Photo Picker or Camera Intent
val r1 = ocr.detectText("content://media/external/images/media/1002").asLines().run()

// 2. Local filesystem path
val r2 = ocr.detectText("/data/user/0/com.example/cache/doc.jpg").asLines().run()

// 3. Encoded image ByteArray
val rawBytes: ByteArray = file.readBytes()
val r3 = ocr.detectText(rawBytes).asLines().run()

// 4. Decoded interleaved RGB8 pixel buffer
val r4 = ocr.detectText()
    .rgb(rgbBytes, width, height, width * 3)
    .asLines()
    .run()
```

## 3. Fluent Builder Chaining (`DetectTextBuilder`)

Configure options dynamically per scan using `.asLines()`, `.asWords()`, or `.asSpatial()` followed by `.run()`:

```kotlin
val result = ocr.detectText("/path/to/invoice.jpg")
    .withAutoUnwarp()
    .withLayoutReading()
    .withPostDetection { copy(useDilation = true, unclipRatio = 2.0f) }
    .withPreRecognition { copy(lineYThreshold = 0.5f, wordXThreshold = 0.4f) }
    .withPostRecognition { copy(scoreThreshold = 0.6f) }
    .asLines()
    .run()
```

## 4. Complete Jetpack Compose Document Scanner

```kotlin
package com.example.ocr

import android.net.Uri
import androidx.activity.compose.rememberLauncherForActivityResult
import androidx.activity.result.contract.ActivityResultContracts
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.text.font.FontFamily
import androidx.compose.ui.unit.dp
import com.byrizki.rusto.*
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext

@Composable
fun DocumentScannerScreen() {
    val context = LocalContext.current
    var selectedUri by remember { mutableStateOf<Uri?>(null) }
    var results by remember { mutableStateOf<List<TextResult>>(emptyList()) }
    var spatialText by remember { mutableStateOf("") }
    var isSpatial by remember { mutableStateOf(false) }
    var isLoading by remember { mutableStateOf(false) }

    val photoPicker = rememberLauncherForActivityResult(
        contract = ActivityResultContracts.PickVisualMedia()
    ) { uri ->
        if (uri != null) {
            selectedUri = uri
        }
    }

    LaunchedEffect(selectedUri, isSpatial) {
        val uri = selectedUri ?: return@LaunchedEffect
        isLoading = true

        withContext(Dispatchers.Default) {
            RustO.initialize(context).use { ocr ->
                if (isSpatial) {
                    val spatialRes = ocr.detectText(uri.toString())
                        .withAutoUnwarp()
                        .asSpatial()
                        .run()
                    if (spatialRes is DetectTextResult.Spatial) {
                        spatialText = spatialRes.text
                        results = emptyList()
                    }
                } else {
                    val structRes = ocr.detectText(uri.toString())
                        .withAutoUnwarp()
                        .withLayoutReading()
                        .asLines()
                        .run()
                    if (structRes is DetectTextResult.Structured) {
                        results = structRes.items
                        spatialText = ""
                    }
                }
            }
        }
        isLoading = false
    }

    Column(modifier = Modifier.fillMaxSize().padding(16.dp)) {
        Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            Button(
                onClick = { photoPicker.launch(androidx.activity.result.PickVisualMediaRequest()) },
                modifier = Modifier.weight(1f)
            ) {
                Text("Select Document")
            }

            FilterChip(
                selected = isSpatial,
                onClick = { isSpatial = !isSpatial },
                label = { Text(if (isSpatial) "Spatial Layout" else "Structured Lines") }
            )
        }

        if (isLoading) {
            CircularProgressIndicator(modifier = Modifier.padding(vertical = 16.dp))
        }

        Spacer(modifier = Modifier.height(16.dp))

        if (isSpatial && spatialText.isNotEmpty()) {
            Text(
                text = spatialText,
                fontFamily = FontFamily.Monospace,
                style = MaterialTheme.typography.bodySmall
            )
        } else {
            LazyColumn(verticalArrangement = Arrangement.spacedBy(8.dp)) {
                items(results) { item ->
                    Card(modifier = Modifier.fillMaxWidth()) {
                        Row(
                            modifier = Modifier.padding(12.dp),
                            horizontalArrangement = Arrangement.SpaceBetween
                        ) {
                            Text(text = item.text, modifier = Modifier.weight(1f))
                            Text(
                                text = "${(item.score * 100).toInt()}%",
                                style = MaterialTheme.typography.bodySmall,
                                color = MaterialTheme.colorScheme.primary
                            )
                        }
                    }
                }
            }
        }
    }
}
```
