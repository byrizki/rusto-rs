Get up and running with RustO! across all 12 SDKs and the `rusto` CLI using the unified v0.4.0 fluent builder API (`as_lines` / `asLines` / `AsLines` + `.run()` / `.Run()`).

## Sample Document Image

Assume we have an input receipt image `receipt.jpg`:

```text
┌────────────────────────────────────────┐
│             COFFEE SHOP                │
│ 2026-08-20                     12:30   │
│                                        │
│ Item                  Qty       Price  │
│ ────────────────────────────────────── │
│ Espresso Double         1       $4.50  │
│ Croissant               2       $7.00  │
│ Flat White              1       $5.00  │
│ ────────────────────────────────────── │
│ Total                          $16.50  │
└────────────────────────────────────────┘
```

## 1. Rust

```rust
use rusto::download::parse_config_with_auto_download;
use rusto::{DetectTextBuilderExt, DetectTextResult, ImageSource, RustO};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    // 1. Register backends & initialize zero-config PP-OCRv6 Tiny FP32 (auto-downloads on first run)
    rusto::init();
    let config = parse_config_with_auto_download("{}")?;
    let mut ocr = RustO::initialize(config)?;

    // 2. Run structured line OCR with fluent builder chaining
    let image = ImageSource::Path("receipt.jpg".into());
    let result = ocr
        .detect_text_builder(image.clone())
        .with_auto_unwarp(true)
        .with_layout_reading(true)
        .as_lines()
        .run()?;

    if let DetectTextResult::Structured(lines) = result {
        for line in lines {
            println!(
                "[{:5.1}, {:5.1}, {:5.1}, {:5.1}] (Score: {:.2}) {}",
                line.frame.left, line.frame.top, line.frame.width, line.frame.height,
                line.score, line.text
            );
        }
    }

    // 3. Run 2D spatial layout reconstruction
    if let DetectTextResult::Spatial(spatial_text) = ocr
        .detect_text_builder(image)
        .as_spatial()
        .run()?
    {
        println!("\n--- Reconstructed Spatial Layout ---\n{spatial_text}");
    }

    Ok(())
}
```

## 4. Golang

```go
package main

import (
	"fmt"
	"log"

	"github.com/byrizki/rusto-rs/packages/go"
)

func main() {
	// Zero-config initialization (auto-downloads PP-OCRv6 Tiny FP32 on first run)
	engine, err := rusto.Initialize()
	if err != nil {
		log.Fatalf("Init failed: %v", err)
	}
	defer engine.Close()

	// Structured line detection
	res, err := engine.DetectText("receipt.jpg").
		WithAutoUnwarp(true).
		WithLayoutReading(true).
		AsLines().
		Run()
	if err != nil {
		log.Fatalf("OCR failed: %v", err)
	}

	for _, line := range res.Results {
		fmt.Printf("[%.2f] %s\n", line.Score, line.Text)
	}

	// Spatial 2D visual layout
	spatialRes, err := engine.DetectText("receipt.jpg").
		AsSpatial().
		Run()
	if err == nil && spatialRes.IsSpatial {
		fmt.Println("--- Spatial Layout ---")
		fmt.Println(spatialRes.Text)
	}
}
```

## 5. Java

```java
import com.byrizki.rusto.RustO;
import com.byrizki.rusto.TextResult;

public class App {
    public static void main(String[] args) {
        try (RustO engine = RustO.initialize()) {
            RustO.DetectTextResult res = engine.detectText("receipt.jpg")
                .withAutoUnwarp()
                .withLayoutReading()
                .asLines()
                .run();

            for (TextResult line : res.results()) {
                System.out.printf("[%.2f] %s%n", line.score(), line.text());
            }

            RustO.DetectTextResult spatial = engine.detectText("receipt.jpg")
                .asSpatial()
                .run();
            System.out.println("--- Spatial Text ---");
            System.out.println(spatial.text());
        }
    }
}
```

## 9. React

```tsx
import { RustoProvider, useOcr } from '@rustors/react';

const models = {
  detection: '/models/ppocrv6-tiny/det.rten',
  recognition: '/models/ppocrv6-tiny/rec.rten',
  dictionary: '/models/ppocrv6-tiny/dict.txt',
};

function Scanner() {
  const { detect, results, isProcessing } = useOcr();

  return (
    <div>
      <input
        type="file"
        accept="image/*"
        onChange={(e) =>
          e.target.files?.[0] &&
          detect(e.target.files[0], { autoUnwarp: true, layoutReading: true })
        }
      />
      {isProcessing && <p>Scanning...</p>}
      {results.map((r, i) => (
        <p key={i}>
          {r.text} ({r.score.toFixed(2)})
        </p>
      ))}
    </div>
  );
}

export function App() {
  return (
    <RustoProvider config={{ preset: 'ppv6', models }}>
      <Scanner />
    </RustoProvider>
  );
}
```

## 10. React Native

```typescript
import React, { useEffect, useState } from 'react';
import { Button, SafeAreaView, ScrollView, Text } from 'react-native';
import { close, detectText, initialize, TextResult } from '@rustors/react-native';

export default function QuickScanApp() {
  const [lines, setLines] = useState<TextResult[]>([]);

  useEffect(() => {
    initialize();
    return () => {
      close();
    };
  }, []);

  const handleScanStructured = async () => {
    const results = await detectText({ uri: 'file:///path/to/receipt.jpg' })
      .withAutoUnwarp()
      .withLayoutReading()
      .asLines()
      .run();
    setLines(results);
  };

  return (
    <SafeAreaView style={{ flex: 1, padding: 16 }}>
      <Button title="Scan Lines" onPress={handleScanStructured} />
      <ScrollView style={{ marginTop: 16 }}>
        {lines.map((line, idx) => (
          <Text key={idx}>{line.text} ({Math.round(line.score * 100)}%)</Text>
        ))}
      </ScrollView>
    </SafeAreaView>
  );
}
```

## 13. Command Line Interface (`rusto`)

```bash
# 1. Zero-config OCR (auto-downloads PP-OCRv6 Tiny FP32 on first run, outputs JSON)
rusto receipt.jpg

# 2. Explicit ocr subcommand with 2D spatial layout reconstruction
rusto ocr receipt.jpg --format spatial

# 3. Pre-download default model and auxiliary stages for offline environments
rusto download --model ppocrv6-tiny --unwarp --layout --reading-order

# 4. Output tab-separated values (TSV) with confidence filter
rusto receipt.jpg --format tsv --score-threshold 0.6
```
